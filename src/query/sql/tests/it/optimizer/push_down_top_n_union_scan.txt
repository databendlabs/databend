    ├── right: [t2.b (#2)]

    ├── Scan
    │   ├── table: default.t1 (#0)
    │   ├── filters: []
    │   ├── order by: [t1.a (#0) ASC]
    │   └── limit: NONE
    └── Scan
        ├── table: default.t2 (#1)
        ├── filters: []
        ├── order by: [t2.b (#2) ASC]
        └── limit: NONE
